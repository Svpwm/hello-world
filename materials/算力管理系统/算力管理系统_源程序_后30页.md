"""

import logging
import uuid
import yaml
from datetime import datetime
from typing import Dict, List, Optional, Any, Tuple

from sqlalchemy import select, update, and_, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.compute import (
    ComputeCluster, ComputeNode, ClusterType, NodeStatus
)
from app.schemas.cluster import ClusterCreate, ClusterUpdate
from app.core.exceptions import ClusterNotFoundException

logger = logging.getLogger(__name__)


class ClusterService:
    """集群服务"""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def create_cluster(self, cluster_data: ClusterCreate) -> ComputeCluster:
        """创建集群"""
        cluster_id = f"cluster_{uuid.uuid4().hex[:12]}"
        
        cluster = ComputeCluster(
            cluster_id=cluster_id,
            name=cluster_data.name,
            description=cluster_data.description,
            cluster_type=ClusterType(cluster_data.cluster_type),
            region=cluster_data.region,
            zone=cluster_data.zone,
            scheduler_type=cluster_data.scheduler_type,
            scheduler_config=cluster_data.scheduler_config or {},
            status="active"
        )
        
        self.db.add(cluster)
        await self.db.commit()
        await self.db.refresh(cluster)
        
        logger.info(f"Created cluster: {cluster_id}")
        return cluster
    
    async def get_cluster(self, cluster_id: str) -> Optional[ComputeCluster]:
        """获取集群"""
        result = await self.db.execute(
            select(ComputeCluster).where(ComputeCluster.cluster_id == cluster_id)
        )
        return result.scalar_one_or_none()
    
    async def list_clusters(
        self,
        region: Optional[str] = None,
        cluster_type: Optional[str] = None,
        skip: int = 0,
        limit: int = 20
    ) -> Tuple[List[ComputeCluster], int]:
        """获取集群列表"""
        query = select(ComputeCluster)
        count_query = select(func.count(ComputeCluster.id))
        
        if region:
            query = query.where(ComputeCluster.region == region)
            count_query = count_query.where(ComputeCluster.region == region)
        
        if cluster_type:
            query = query.where(ComputeCluster.cluster_type == cluster_type)
            count_query = count_query.where(ComputeCluster.cluster_type == cluster_type)
        
        query = query.offset(skip).limit(limit)
        
        result = await self.db.execute(query)
        clusters = list(result.scalars().all())
        
        count_result = await self.db.execute(count_query)
        total = count_result.scalar()
        
        return clusters, total

    async def update_cluster(
        self,
        cluster_id: str,
        update_data: ClusterUpdate
    ) -> ComputeCluster:
        """更新集群"""
        cluster = await self.get_cluster(cluster_id)
        
        if not cluster:
            raise ClusterNotFoundException(f"Cluster not found: {cluster_id}")
        
        if update_data.name is not None:
            cluster.name = update_data.name
        if update_data.description is not None:
            cluster.description = update_data.description
        if update_data.scheduler_config is not None:
            cluster.scheduler_config = update_data.scheduler_config
        
        await self.db.commit()
        return cluster
    
    async def delete_cluster(self, cluster_id: str) -> bool:
        """删除集群"""
        cluster = await self.get_cluster(cluster_id)
        
        if not cluster:
            raise ClusterNotFoundException(f"Cluster not found: {cluster_id}")
        
        nodes_result = await self.db.execute(
            select(ComputeNode).where(ComputeNode.cluster_id == cluster_id)
        )
        nodes = nodes_result.scalars().all()
        
        if nodes:
            raise ValueError("Cannot delete cluster with registered nodes")
        
        await self.db.delete(cluster)
        await self.db.commit()
        
        logger.info(f"Deleted cluster: {cluster_id}")
        return True
    
    async def get_cluster_resources(self, cluster_id: str) -> Dict[str, Any]:
        """获取集群资源汇总"""
        result = await self.db.execute(
            select(
                func.count(ComputeNode.id),
                func.sum(ComputeNode.cpu_cores),
                func.sum(ComputeNode.memory_gb),
                func.sum(ComputeNode.gpu_count),
                func.sum(ComputeNode.allocated_cpu),
                func.sum(ComputeNode.allocated_memory_gb),
                func.sum(ComputeNode.allocated_gpus)
            )
            .where(
                and_(
                    ComputeNode.cluster_id == cluster_id,
                    ComputeNode.status == NodeStatus.ONLINE
                )
            )
        )
        
        row = result.one()
        node_count, total_cpu, total_memory, total_gpu, \
        alloc_cpu, alloc_memory, alloc_gpu = row
        
        return {
            "cluster_id": cluster_id,
            "online_nodes": node_count or 0,
            "cpu": {
                "total": float(total_cpu or 0),
                "allocated": float(alloc_cpu or 0),
                "available": float((total_cpu or 0) - (alloc_cpu or 0))
            },
            "memory_gb": {
                "total": float(total_memory or 0),
                "allocated": float(alloc_memory or 0),
                "available": float((total_memory or 0) - (alloc_memory or 0))

                "total": int(total_gpu or 0),

        cluster = await self.cluster_service.create_cluster(cluster_data)
        
        for node in nodes.items:
            await self._register_k8s_node(cluster.cluster_id, node)
        
        return cluster

"""

import logging
import asyncio
from datetime import datetime, timedelta
from typing import Dict, List, Optional, Any
from dataclasses import dataclass, field
from enum import Enum
import aiohttp
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

from sqlalchemy import select, update, and_
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings

logger = logging.getLogger(__name__)


class AlertLevel(str, Enum):
    """告警级别"""
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"


@dataclass
class Alert:
    """告警"""
    alert_id: str
    level: AlertLevel
    title: str
    message: str
    labels: Dict[str, str] = field(default_factory=dict)
    timestamp: datetime = field(default_factory=datetime.utcnow)
    resolved: bool = False
    resolved_at: Optional[datetime] = None


class AlertService:
    """告警服务"""
    
    def __init__(self):
        self._alerts: Dict[str, Alert] = {}
        self._alert_history: List[Alert] = []
        self._suppression_window = timedelta(minutes=10)
        self._last_sent: Dict[str, datetime] = {}
    
    async def send_alert(
        self,
        level: str,
        title: str,
        message: str,
        labels: Optional[Dict[str, str]] = None
    ) -> None:
        """发送告警"""
        alert_key = self._generate_alert_key(title, labels or {})
        
        if self._is_suppressed(alert_key):
            return
        
        alert = Alert(
            alert_id=f"alert_{datetime.utcnow().strftime('%Y%m%d%H%M%S')}",
            level=AlertLevel(level),
            title=title,
            message=message,
            labels=labels or {}
        )
        
        self._alerts[alert_key] = alert
        self._alert_history.append(alert)
        self._last_sent[alert_key] = datetime.utcnow()
        
        await self._dispatch_alert(alert)
        
        logger.info(f"Alert sent: {title}")
    
    def _generate_alert_key(self, title: str, labels: Dict[str, str]) -> str:
        """生成告警键"""
        label_str = "_".join(f"{k}={v}" for k, v in sorted(labels.items()))
        return f"{title}_{label_str}"
    
    def _is_suppressed(self, alert_key: str) -> bool:
        """检查是否被抑制"""
        last_time = self._last_sent.get(alert_key)
        if not last_time:
            return False
        return datetime.utcnow() - last_time < self._suppression_window

    async def _dispatch_alert(self, alert: Alert) -> None:
        """分发告警"""
        tasks = []
        
        if settings.ALERT_WEBHOOK_URL:
            tasks.append(self._send_webhook(alert))
        
        if settings.ALERT_EMAIL_ENABLED:
            tasks.append(self._send_email(alert))
        
        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)
    
    async def _send_webhook(self, alert: Alert) -> None:
        """发送Webhook"""
        payload = {
            "alert_id": alert.alert_id,
            "level": alert.level.value,
            "title": alert.title,
            "message": alert.message,
            "labels": alert.labels,
            "timestamp": alert.timestamp.isoformat()
        }
        
        async with aiohttp.ClientSession() as session:
            try:
                async with session.post(
                    settings.ALERT_WEBHOOK_URL,
                    json=payload,
                    timeout=30
                ) as resp:
                    if resp.status != 200:
                        logger.warning(f"Webhook failed: {resp.status}")
            except Exception as e:
                logger.error(f"Webhook error: {e}")
    
    async def _send_email(self, alert: Alert) -> None:
        """发送邮件"""
        subject = f"[{alert.level.value.upper()}] {alert.title}"
        
        body = f"""
        告警级别: {alert.level.value}
        告警标题: {alert.title}
        告警内容: {alert.message}
        告警时间: {alert.timestamp}
        标签: {alert.labels}
        """
        
        msg = MIMEMultipart()
        msg["Subject"] = subject
        msg["From"] = settings.SMTP_FROM
        msg["To"] = settings.ALERT_EMAIL_TO
        
        msg.attach(MIMEText(body, "plain"))
        
        try:
            with smtplib.SMTP(settings.SMTP_HOST, settings.SMTP_PORT) as server:
                server.starttls()
                server.login(settings.SMTP_USER, settings.SMTP_PASSWORD)
                server.send_message(msg)
        except Exception as e:
            logger.error(f"Email send error: {e}")
    
    async def resolve_alert(self, alert_key: str) -> None:
        """解决告警"""
        alert = self._alerts.get(alert_key)
        if alert:
            alert.resolved = True
            alert.resolved_at = datetime.utcnow()
            logger.info(f"Alert resolved: {alert.title}")
    
    def get_active_alerts(self) -> List[Alert]:
        """获取活跃告警"""
        return [a for a in self._alerts.values() if not a.resolved]
    
    def get_alert_history(
        self,
        since: Optional[datetime] = None,
        limit: int = 100
    ) -> List[Alert]:
        """获取告警历史"""
        alerts = self._alert_history
        
        if since:
            alerts = [a for a in alerts if a.timestamp >= since]
        
        return alerts[-limit:]

) -> User:
    """获取当前用户"""

        payload = jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM]
        )

                status_code=status.HTTP_401_UNAUTHORIZED,

            )
        
        from sqlalchemy import select

        if not user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="User not found"
            )
        
        if not user.is_active:
            raise HTTPException(

from dataclasses import dataclass

from app.core.config import settings

@dataclass
class QueueMessage:
    """队列消息"""
    topic: str
    payload: Dict[str, Any]
    timestamp: float


class MessageQueue:
    """消息队列"""
    
    def __init__(self):
        self._connection = None
        self._channel = None
        self._exchange = None
        self._subscribers: Dict[str, List[Callable]] = {}
    
    async def connect(self) -> None:
        """连接消息队列"""
        self._connection = await aio_pika.connect_robust(settings.RABBITMQ_URL)
        self._channel = await self._connection.channel()
        
        self._exchange = await self._channel.declare_exchange(
            "compute_events",
            ExchangeType.TOPIC,
            durable=True
        )
        
        logger.info("Message queue connected")
    
    async def disconnect(self) -> None:
        """断开连接"""
        if self._connection:
            await self._connection.close()
            logger.info("Message queue disconnected")
    
    async def publish(self, topic: str, payload: Dict[str, Any]) -> None:
        """发布消息"""
        if not self._exchange:
            return
        
        message = Message(
            body=json.dumps(payload).encode(),
            content_type="application/json"
        )
        
        await self._exchange.publish(message, routing_key=topic)
        logger.debug(f"Published message to {topic}")
    
    async def subscribe(
        self,
        topic: str,
        callback: Callable[[Dict[str, Any]], None]
    ) -> None:
        """订阅消息"""
        if not self._channel:
            return
        
        queue = await self._channel.declare_queue(exclusive=True)
        await queue.bind(self._exchange, routing_key=topic)
        
        async def on_message(message):
            async with message.process():
                payload = json.loads(message.body.decode())
                await callback(payload)
        
        await queue.consume(on_message)
        
        if topic not in self._subscribers:
            self._subscribers[topic] = []
        self._subscribers[topic].append(callback)
        
        logger.info(f"Subscribed to {topic}")
