    async def start(self) -> None:
        """启动调度器"""
        self._running = True
        asyncio.create_task(self._scheduling_loop())
        logger.info("Compute scheduler started")
    
    async def stop(self) -> None:
        """停止调度器"""
        self._running = False
        logger.info("Compute scheduler stopped")
    
    async def _scheduling_loop(self) -> None:
        """调度循环"""
        while self._running:
            try:
                await self._run_scheduling_cycle()
            except Exception as e:
                logger.error(f"Scheduling cycle error: {e}")
            
            await asyncio.sleep(self._scheduling_interval)
    
    async def _run_scheduling_cycle(self) -> None:
        """运行调度周期"""
        pending_tasks = await self._get_pending_tasks()
        
        if not pending_tasks:
            return
        
        sorted_tasks = self._sort_tasks(pending_tasks)
        
        for task in sorted_tasks:
            request = self._build_scheduling_request(task)
            result = await self._try_schedule(request)
            
            if result.success:
                await self._assign_task(task, result)
            else:
                logger.debug(f"Task {task.task_id} could not be scheduled: {result.message}")
    
    async def _get_pending_tasks(self) -> List[ComputeTask]:
        """获取待调度任务"""
        result = await self.db.execute(
            select(ComputeTask)
            .where(ComputeTask.status == TaskStatus.PENDING)
            .order_by(ComputeTask.priority.desc(), ComputeTask.submit_time)
            .limit(100)
        )
        return list(result.scalars().all())
    
    def _sort_tasks(self, tasks: List[ComputeTask]) -> List[ComputeTask]:
        """排序任务"""
        if self._policy == SchedulingPolicy.FIFO:
            return sorted(tasks, key=lambda t: t.submit_time)
        elif self._policy == SchedulingPolicy.PRIORITY:
            return sorted(tasks, key=lambda t: (-t.priority, t.submit_time))
        elif self._policy == SchedulingPolicy.FAIR_SHARE:
            return self._fair_share_sort(tasks)
        return tasks
    
    def _fair_share_sort(self, tasks: List[ComputeTask]) -> List[ComputeTask]:
        """公平份额排序"""
        user_counts = defaultdict(int)
        for task in tasks:
            user_counts[task.user_id] += 1
        
        return sorted(tasks, key=lambda t: (user_counts[t.user_id], t.submit_time))

    def _build_scheduling_request(self, task: ComputeTask) -> SchedulingRequest:
        """构建调度请求"""
        return SchedulingRequest(
            task_id=task.task_id,
            cpu_request=task.cpu_request,
            memory_request_gb=task.memory_request_gb,
            gpu_request=task.gpu_request,
            gpu_type=task.gpu_type,
            priority=task.priority,
            preemptible=task.preemptible,
            node_selector=task.metadata.get("node_selector", {}),
            tolerations=task.metadata.get("tolerations", []),
            affinity=task.metadata.get("affinity")
        )
    
    async def _try_schedule(self, request: SchedulingRequest) -> SchedulingResult:
        """尝试调度"""
        candidates = await self._find_candidate_nodes(request)
        
        if not candidates:
            return SchedulingResult(
                success=False,
                message="No suitable nodes available"
            )
        
        best_node = self._select_best_node(candidates, request)
        
        if not best_node:
            return SchedulingResult(
                success=False,
                message="No node meets requirements"
            )
        
        if request.gpu_request > 0:
            gpu_ids = await self._allocate_gpus(
                best_node.node_id,
                request.gpu_request,
                request.gpu_type
            )
            
            if len(gpu_ids) < request.gpu_request:
                return SchedulingResult(
                    success=False,
                    message="Insufficient GPUs available"
                )
        else:
            gpu_ids = []
        
        return SchedulingResult(
            success=True,
            node_id=best_node.node_id,
            gpu_ids=gpu_ids,
            message="Scheduled successfully"
        )
    
    async def _find_candidate_nodes(
        self,
        request: SchedulingRequest
    ) -> List[ComputeNode]:
        """查找候选节点"""
        query = select(ComputeNode).where(
            and_(
                ComputeNode.status == NodeStatus.ONLINE,
                ComputeNode.cpu_cores - ComputeNode.allocated_cpu >= request.cpu_request,
                ComputeNode.memory_gb - ComputeNode.allocated_memory_gb >= request.memory_request_gb
            )
        )
        
        if request.gpu_request > 0:
            query = query.where(
                ComputeNode.gpu_count - ComputeNode.allocated_gpus >= request.gpu_request
            )
            
            if request.gpu_type:
                query = query.where(ComputeNode.gpu_type == request.gpu_type)
        
        result = await self.db.execute(query)
        candidates = list(result.scalars().all())
        
        filtered = self._filter_by_selector(candidates, request.node_selector)
        filtered = self._filter_by_taints(filtered, request.tolerations)

        self,
        nodes: List[ComputeNode],
        tolerations: List[Dict]
    ) -> List[ComputeNode]:
        """根据污点过滤节点"""
        result = []
        
        for node in nodes:
            taints = node.taints or []
            
            if not taints:
                result.append(node)
                continue
            
            tolerated = True
            for taint in taints:
                if not self._is_tolerated(taint, tolerations):
                    tolerated = False
                    break
            
            if tolerated:
                result.append(node)
        
        return result

        """检查污点是否被容忍"""
        for toleration in tolerations:
            if toleration.get("key") == taint.get("key"):
                if toleration.get("operator") == "Exists":
                    return True
                if toleration.get("value") == taint.get("value"):
                    return True
        return False

    def _best_fit_selection(
        self,
        nodes: List[ComputeNode],
        request: SchedulingRequest
    ) -> ComputeNode:
        """最佳适应选择"""
        def score(node: ComputeNode) -> float:
            available_cpu = node.cpu_cores - node.allocated_cpu
            available_mem = node.memory_gb - node.allocated_memory_gb
            
            cpu_fit = request.cpu_request / available_cpu if available_cpu > 0 else 0
            mem_fit = request.memory_request_gb / available_mem if available_mem > 0 else 0
            
            return (cpu_fit + mem_fit) / 2
        
        return max(nodes, key=score)
    
    def _spread_selection(self, nodes: List[ComputeNode]) -> ComputeNode:
        """分散选择"""
        def utilization(node: ComputeNode) -> float:
            cpu_util = node.allocated_cpu / node.cpu_cores if node.cpu_cores > 0 else 1
            mem_util = node.allocated_memory_gb / node.memory_gb if node.memory_gb > 0 else 1
            return (cpu_util + mem_util) / 2
        
        return min(nodes, key=utilization)

    async def _process_billing(self) -> None:
        """处理计费"""
        result = await self.db.execute(
            select(ComputeTask).where(
                and_(
                    ComputeTask.status.in_([TaskStatus.COMPLETED, TaskStatus.FAILED, TaskStatus.TIMEOUT]),
                    ComputeTask.end_time.isnot(None)
                )
            )
        )
        
        tasks = result.scalars().all()
        
        for task in tasks:
            existing = await self.db.execute(
                select(BillingRecord).where(BillingRecord.task_id == task.task_id)
            )
            if existing.scalar_one_or_none():
                continue
            
            records = await self._generate_billing_records(task)
            
            for record in records:
                self.db.add(record)
            
            await self.db.commit()
            
            logger.info(f"Generated {len(records)} billing records for task {task.task_id}")
    
    async def _generate_billing_records(
        self,
        task: ComputeTask
    ) -> List[BillingRecord]:
        """生成计费记录"""
        records = []
        
        if not task.start_time or not task.end_time:
            return records
        
        duration_hours = (task.end_time - task.start_time).total_seconds() / 3600
        
        if task.gpu_request and task.gpu_request > 0:
            gpu_type = task.gpu_type or "gpu_a100_80g"
            pricing = self.PRICING_RULES.get(gpu_type)
            
            if pricing:
                gpu_usage = duration_hours * task.gpu_request
                gpu_cost = gpu_usage * pricing.unit_price
                
                records.append(BillingRecord(
                    record_id=f"bill_{uuid.uuid4().hex[:12]}",
                    tenant_id=task.tenant_id or task.user_id,
                    user_id=task.user_id,
                    task_id=task.task_id,
                    resource_type=gpu_type,
                    usage_amount=float(gpu_usage),
                    usage_unit="gpu_hour",
                    unit_price=float(pricing.unit_price),
                    currency=pricing.currency,
                    total_cost=float(gpu_cost),
                    start_time=task.start_time,
                    end_time=task.end_time,
                    status="completed"
                ))
        
        if task.cpu_request and task.cpu_request > 0:
            pricing = self.PRICING_RULES["cpu_core"]
            cpu_usage = duration_hours * task.cpu_request
            cpu_cost = cpu_usage * pricing.unit_price
            
            records.append(BillingRecord(
                record_id=f"bill_{uuid.uuid4().hex[:12]}",
