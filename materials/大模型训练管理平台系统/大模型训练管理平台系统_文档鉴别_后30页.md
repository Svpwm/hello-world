## 第31页

#### 6.2 训练任务接口

##### 6.2.1 创建训练任务

```yaml
POST /api/v1/training/jobs

Request:
  Headers:
    Authorization: Bearer <token>
    Content-Type: application/json
  
  Query Parameters:
    project_id: string (required)  # 项目ID
  
  Body:
    name: string (required)         # 任务名称
    description: string             # 任务描述
    model_config:                   # 模型配置
      model_type: string            # 模型类型
      model_name: string            # 预训练模型名称
      hidden_size: integer          # 隐藏层大小
      num_layers: integer           # 层数
    training_config:                # 训练配置
      num_epochs: integer           # 训练轮数
      batch_size: integer           # 批次大小
      learning_rate: float          # 学习率
      optimizer: string             # 优化器
      scheduler: string             # 学习率调度器
      dataset:                      # 数据集配置
        train_path: string          # 训练集路径
        val_path: string            # 验证集路径
    resource_config:                # 资源配置
      gpu_count: integer            # GPU数量
      gpu_type: string              # GPU类型
      node_count: integer           # 节点数量
    priority: integer               # 优先级 (0-100)

Response:
  Success (201):
    {
      "code": 0,
      "message": "success",
      "data": {
        "job_id": "job_abc123",
        "project_id": "proj_001",
        "name": "LLaMA微调任务",
        "status": "pending",
        "created_at": "2024-03-15T10:00:00Z"
      }
    }
  
  Error (400):
    {
      "code": 40001,
      "message": "参数验证失败",
      "errors": [
        {"field": "gpu_count", "message": "超出配额限制"}
      ]
    }
```

### 7. 安全设计

#### 7.1 认证授权

##### 7.1.1 认证方式

系统支持多种认证方式：

| 认证方式 | 说明 | 适用场景 |
|---------|------|---------|
| JWT Token | 基于JWT的令牌认证 | Web/API访问 |
| API Key | 长期有效的API密钥 | 服务间调用 |
| OAuth 2.0 | 第三方登录 | 企业SSO集成 |
| LDAP | 目录服务认证 | 企业用户管理 |

##### 7.1.2 JWT Token结构

```json
{
  "header": {
    "alg": "RS256",
    "typ": "JWT"
  },
  "payload": {
    "sub": "user_001",
    "username": "zhangsan",
    "email": "zhangsan@example.com",
    "roles": ["algorithm_engineer"],
    "permissions": ["training:create", "training:read"],
    "projects": ["proj_001", "proj_002"],
    "iat": 1710489600,
    "exp": 1710576000,
    "jti": "token_unique_id"
  },
  "signature": "..."

```

#### 7.2 数据安全

##### 7.2.1 数据加密

| 场景 | 加密方式 | 说明 |
|------|---------|------|
| 传输加密 | TLS 1.3 | 所有API通信使用HTTPS |
| 存储加密 | AES-256-GCM | 敏感数据加密存储 |
| 密码存储 | bcrypt | 用户密码哈希存储 |
| Token签名 | RS256 | JWT使用RSA签名 |
| 模型加密 | AES-256 | 可选的模型文件加密 |

```bash
    echo "All services stopped"

    echo "API Service:"
    systemctl status llm-platform-api --no-pager | head -5
    echo ""
    echo "Worker Service:"
    systemctl status llm-platform-worker --no-pager | head -5
    echo ""
    echo "Scheduler Service:"
    systemctl status llm-platform-scheduler --no-pager | head -5
}
```

```bash
#!/bin/bash
# gpu_manager.sh - GPU管理脚本

# 查看GPU状态
gpu_status() {
    echo "=== GPU Status ==="
    nvidia-smi --query-gpu=index,name,utilization.gpu,memory.used,memory.total,temperature.gpu \
        --format=csv,noheader,nounits | \
    while IFS=',' read -r idx name util mem_used mem_total temp; do
        printf "GPU %s (%s): Util=%s%% Mem=%s/%sMB Temp=%s°C\n" \
            "$idx" "$name" "$util" "$mem_used" "$mem_total" "$temp"
    done
}

# 查看GPU进程
gpu_processes() {
    echo "=== GPU Processes ==="
    nvidia-smi --query-compute-apps=pid,process_name,gpu_uuid,used_memory \
        --format=csv,noheader
}

# 清理僵尸进程
cleanup_zombie() {
    echo "Cleaning up zombie GPU processes..."
    
    # 获取所有占用GPU的进程
    nvidia-smi --query-compute-apps=pid --format=csv,noheader | while read pid; do
        # 检查进程是否存在
        if ! kill -0 "$pid" 2>/dev/null; then
            echo "Found zombie process: $pid"
            # 尝试强制清理
            nvidia-smi --id=0 --compute-reset
        fi
    done
    
    echo "Cleanup completed"
}

# GPU压力测试
stress_test() {
    gpu_id=${1:-0}
    duration=${2:-60}
    
    echo "Running GPU stress test on GPU $gpu_id for ${duration}s..."
    python -c "
import torch
import time
device = torch.device('cuda:$gpu_id')
x = torch.randn(10000, 10000, device=device)
start = time.time()
while time.time() - start < $duration:
    y = torch.matmul(x, x)
print('Stress test completed')
"
}

case "$1" in
    status)    gpu_status ;;
    processes) gpu_processes ;;
    cleanup)   cleanup_zombie ;;
    stress)    stress_test "$2" "$3" ;;
    *)         echo "Usage: $0 {status|processes|cleanup|stress}" ;;
esac
```

```bash
#!/bin/bash
# backup.sh - 备份脚本

BACKUP_DIR="/backup/llm-platform"
DATE=$(date +%Y%m%d)
RETENTION_DAYS=30

# 数据库备份
backup_database() {
    echo "Backing up database..."
    
    DUMP_FILE="${BACKUP_DIR}/db/llm_platform_${DATE}.sql.gz"
    
    pg_dump -h localhost -U postgres llm_platform | gzip > "$DUMP_FILE"
    
    if [ $? -eq 0 ]; then
        echo "Database backup completed: $DUMP_FILE"
    else
        echo "Database backup FAILED"
        return 1
    fi
}

# 配置备份
backup_config() {
    echo "Backing up configurations..."
    
    CONFIG_BACKUP="${BACKUP_DIR}/config/config_${DATE}.tar.gz"
    
    tar -czf "$CONFIG_BACKUP" \
        /etc/llm-platform/ \
        /opt/llm-platform/config/
    
    echo "Config backup completed: $CONFIG_BACKUP"
}

# 模型备份
backup_models() {
    echo "Backing up model repository..."
    
    rsync -av --progress \
        /data/model-repository/ \
        "${BACKUP_DIR}/models/"
    
    echo "Model backup completed"
}

# 清理过期备份
cleanup_old_backups() {
    echo "Cleaning up old backups..."
    
    find "${BACKUP_DIR}/db" -name "*.sql.gz" -mtime +${RETENTION_DAYS} -delete
    find "${BACKUP_DIR}/config" -name "*.tar.gz" -mtime +${RETENTION_DAYS} -delete
    
    echo "Cleanup completed"
}

# 主函数
main() {
    mkdir -p "${BACKUP_DIR}"/{db,config,models}
    
    backup_database
    backup_config
    backup_models
    cleanup_old_backups
    
    echo "=== Backup Summary ==="
    du -sh "${BACKUP_DIR}"/*
}

main
```

```python
scaler = GradScaler()

for batch in dataloader:
    optimizer.zero_grad()
    
    # 前向传播使用FP16
    with autocast():
        outputs = model(batch)
        loss = criterion(outputs, targets)
    
    # 反向传播使用缩放的梯度
    scaler.scale(loss).backward()
    
    # 更新参数
    scaler.step(optimizer)
    scaler.update()
```

```python
from torch.utils.checkpoint import checkpoint

class TransformerBlock(nn.Module):
    def forward(self, x):
        # 使用检查点包装，牺牲计算换显存
        return checkpoint(self._forward, x)
    
    def _forward(self, x):
        x = self.attention(x)
        x = self.feedforward(x)
        return x
```

```python
# Redis缓存策略
class CacheService:
    def __init__(self, redis_client):
        self.redis = redis_client
        
        # 缓存配置
        self.ttl_config = {
            "job_status": 10,      # 任务状态：10秒
            "cluster_status": 30,  # 集群状态：30秒
            "user_info": 300,      # 用户信息：5分钟
            "project_info": 600,   # 项目信息：10分钟
        }
    
    async def get_job_status(self, job_id: str):
        cache_key = f"job_status:{job_id}"
        
        # 尝试从缓存获取
        cached = await self.redis.get(cache_key)
        if cached:
            return json.loads(cached)
        
        # 查询数据库
        status = await self.db.get_job_status(job_id)
        
        # 写入缓存
        await self.redis.setex(
            cache_key, 
            self.ttl_config["job_status"],
            json.dumps(status)
        )
        
        return status
```
