    """应用生命周期管理"""

    logger.info("Shutting down LLM Training Platform...")
    await scheduler.stop()
    await monitor.stop()
    await close_db()
    logger.info("LLM Training Platform stopped")

                logger.info(f"Job {job.job_id} waiting for resources")
                
        except ResourceNotAvailableException as e:
            logger.warning(f"Resources not available for job {job.job_id}: {e}")

    async def start_job(self, job_id: str) -> TrainingJob:
        """启动训练任务"""
        job = await self._get_job(job_id)
        
        if job.status not in [JobStatus.PENDING, JobStatus.QUEUED, JobStatus.PAUSED]:
            raise InvalidJobStateException(
                f"Cannot start job in state: {job.status}"
            )
        
        resources = await self.resource_manager.allocate_resources(
            job_id=job_id,
            gpu_count=job.gpu_count,
            gpu_type=job.gpu_type,
            node_count=job.node_count
        )
        
        if not resources:
            raise ResourceNotAvailableException("No resources available")
        
        job.status = JobStatus.RUNNING
        job.start_time = datetime.utcnow()
        
        await self.db.commit()
        
        await self._launch_training_process(job, resources)
        
        logger.info(f"Started training job: {job_id}")
        
        await self.notification.send_job_notification(
            job, "started", f"Training job {job.name} has started"
        )
        
        return job
    
    async def _launch_training_process(
        self,
        job: TrainingJob,
        resources: Dict[str, Any]
    ) -> None:
        """启动训练进程"""
        from app.services.executor import TrainingExecutor
        
        executor = TrainingExecutor(
            job_id=job.job_id,
            model_config=job.model_config,
            training_config=job.training_config,
            resources=resources,
            output_path=job.output_path,
            checkpoint_path=job.checkpoint_path
        )
        
        asyncio.create_task(executor.run())
    
    async def pause_job(self, job_id: str) -> TrainingJob:
        """暂停训练任务"""
        job = await self._get_job(job_id)
        
        if job.status != JobStatus.RUNNING:
            raise InvalidJobStateException("Can only pause running jobs")
        
        await self._send_control_signal(job_id, "pause")
        
        job.status = JobStatus.PAUSED
        await self.db.commit()
        
        logger.info(f"Paused training job: {job_id}")

    async def resume_job(self, job_id: str) -> TrainingJob:
        """恢复训练任务"""
        job = await self._get_job(job_id)
        
        if job.status != JobStatus.PAUSED:
            raise InvalidJobStateException("Can only resume paused jobs")
        
        latest_checkpoint = await self._get_latest_checkpoint(job_id)
        
        if latest_checkpoint:
            await self._send_control_signal(
                job_id, "resume", 
                {"checkpoint_path": latest_checkpoint.path}
            )
        else:
            await self._send_control_signal(job_id, "resume")
        
        job.status = JobStatus.RUNNING
        await self.db.commit()
        
        logger.info(f"Resumed training job: {job_id}")
        return job
    
    async def cancel_job(self, job_id: str) -> TrainingJob:
        """取消训练任务"""
        job = await self._get_job(job_id)
        
        if job.status in [JobStatus.COMPLETED, JobStatus.CANCELLED]:
            raise InvalidJobStateException(
                f"Cannot cancel job in state: {job.status}"
            )
        
        if job.status == JobStatus.RUNNING:
            await self._send_control_signal(job_id, "stop")
        
        await self.resource_manager.release_resources(job_id)
        
        job.status = JobStatus.CANCELLED
        job.end_time = datetime.utcnow()
        if job.start_time:
            job.elapsed_seconds = int(
                (job.end_time - job.start_time).total_seconds()
            )
        
        await self.db.commit()
        
        logger.info(f"Cancelled training job: {job_id}")
        
        await self.notification.send_job_notification(
            job, "cancelled", f"Training job {job.name} has been cancelled"
        )
        
        return job
    
    async def _send_control_signal(
        self,
        job_id: str,
        signal: str,
        params: Optional[Dict] = None
    ) -> None:
        """发送控制信号"""
        from app.services.message_queue import MessageQueue
        mq = MessageQueue()
        await mq.publish(
            f"job.{job_id}.control",
            {"signal": signal, "params": params or {}}
        )

    async def allocate_resources(

                job_id=job_id,
                gpu_ids=gpu_ids,
                node_names=node_names

            self._allocations[job_id] = allocation
            
            logger.info(f"Allocated {gpu_count} GPUs to job {job_id}")
            
            return {
                "job_id": job_id,
                "gpu_ids": gpu_ids,
                "node_names": node_names,
                "gpu_count": len(gpu_ids)
            }
    
    async def release_resources(self, job_id: str) -> bool:
        """释放资源"""
        async with self._lock:
            allocation = self._allocations.pop(job_id, None)
            
            if not allocation:
                logger.warning(f"No allocation found for job {job_id}")
                return False
            
            await self.db.execute(
                update(GPUResource)
                .where(GPUResource.resource_id.in_(allocation.gpu_ids))
                .values(
                    status=ResourceStatus.AVAILABLE,
                    current_job_id=None
                )
            )
            await self.db.commit()
            
            logger.info(f"Released {allocation.gpu_count} GPUs from job {job_id}")
            
            await self._process_pending_requests()
            
            return True

                logger.info(f"Job {self.config.job_id} paused")

                self._running = False

    async def _train_epoch(
        self,
        model,
        optimizer,
        scheduler,
        train_loader,
        epoch: int
    ) -> None:
        """训练一个epoch"""
        model.train()
        total_loss = 0.0
        step_count = 0
        
        log_interval = self.config.training_config.get("log_interval", 10)
        gradient_accumulation = self.config.training_config.get(
            "gradient_accumulation_steps", 1
        )
        max_grad_norm = self.config.training_config.get("max_grad_norm", 1.0)
        
        for batch_idx, batch in enumerate(train_loader):
            if not self._running:
                break
            
            while self._paused:
                await asyncio.sleep(1)
            
            if torch.cuda.is_available():
                batch = {k: v.cuda() for k, v in batch.items()}
            
            outputs = model(**batch)
            loss = outputs.loss / gradient_accumulation
            
            loss.backward()
            
            if (batch_idx + 1) % gradient_accumulation == 0:
                if max_grad_norm > 0:
                    torch.nn.utils.clip_grad_norm_(
                        model.parameters(), max_grad_norm
                    )
                
                optimizer.step()
                if scheduler:
                    scheduler.step()
                optimizer.zero_grad()
                
                self._current_step += 1
            
            total_loss += loss.item() * gradient_accumulation
            step_count += 1
            
            if self._current_step % log_interval == 0:
                avg_loss = total_loss / step_count
                
                gpu_memory = 0
                gpu_util = 0
                if torch.cuda.is_available():
                    gpu_memory = torch.cuda.memory_allocated() / 1e9
                    gpu_util = torch.cuda.utilization() if hasattr(torch.cuda, 'utilization') else 0
                
                await self._report_metrics({
                    "epoch": epoch,
                    "step": self._current_step,
                    "loss": avg_loss,
                    "learning_rate": optimizer.param_groups[0]["lr"],
                    "gpu_memory_used": gpu_memory,
                    "gpu_utilization": gpu_util
                })
                
                logger.info(
                    f"Epoch {epoch} Step {self._current_step}: "
                    f"loss={avg_loss:.4f}, lr={optimizer.param_groups[0]['lr']:.2e}"
                )

                        mask = labels != -100
                        total_correct += (predictions == labels)[mask].sum().item()
                        total_samples += mask.sum().item()
        
        avg_loss = total_loss / len(val_loader)
        accuracy = total_correct / total_samples if total_samples > 0 else 0
        perplexity = torch.exp(torch.tensor(avg_loss)).item()
        
        model.train()
        
        return {
            "val_loss": avg_loss,
            "val_accuracy": accuracy,
            "val_perplexity": perplexity
        }

        self,

        process = subprocess.Popen(
            cmd,

            stderr=subprocess.PIPE

    def _compute_perplexity(self, losses: List[float], **kwargs) -> float:
        """计算困惑度"""
        if not losses:
            return float("inf")
        avg_loss = np.mean(losses)
        return np.exp(avg_loss)
    
    def _compute_accuracy(
        self,
        predictions: List,
        labels: List,
        **kwargs
    ) -> float:
        """计算准确率"""
        if not predictions or not labels:
            return 0.0
        
        correct = sum(
            1 for p, l in zip(predictions, labels)
            if p == l and l != -100
        )
        total = sum(1 for l in labels if l != -100)
        
        return correct / total if total > 0 else 0.0
    
    def _compute_bleu(self, predictions: List, labels: List, **kwargs) -> float:
        """计算BLEU分数"""
        from nltk.translate.bleu_score import sentence_bleu, SmoothingFunction
        
        scores = []
        smooth = SmoothingFunction().method1
        
        for pred, ref in zip(predictions, labels):
            pred_tokens = self.tokenizer.decode(pred, skip_special_tokens=True).split()
            ref_tokens = self.tokenizer.decode(ref, skip_special_tokens=True).split()
            score = sentence_bleu([ref_tokens], pred_tokens, smoothing_function=smooth)
            scores.append(score)
        
        return np.mean(scores) if scores else 0.0
