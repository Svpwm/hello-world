```python
    def _compute_rouge(self, predictions: List, labels: List, **kwargs) -> Dict[str, float]:
        """计算ROUGE分数"""
        from rouge_score import rouge_scorer
        
        scorer = rouge_scorer.RougeScorer(['rouge1', 'rouge2', 'rougeL'], use_stemmer=True)
        
        rouge_scores = defaultdict(list)
        
        for pred, ref in zip(predictions, labels):
            pred_text = self.tokenizer.decode(pred, skip_special_tokens=True)
            ref_text = self.tokenizer.decode(ref, skip_special_tokens=True)
            
            scores = scorer.score(ref_text, pred_text)
            for key, value in scores.items():
                rouge_scores[key].append(value.fmeasure)
        
        return {
            key: np.mean(values) for key, values in rouge_scores.items()
        }
    
    def _compute_f1(self, predictions: List, labels: List, **kwargs) -> float:
        """计算F1分数"""
        from sklearn.metrics import f1_score
        
        flat_preds = [p for pred in predictions for p in pred if p != -100]
        flat_labels = [l for label in labels for l in label if l != -100]
        
        if not flat_preds or not flat_labels:
            return 0.0
        
        return f1_score(flat_labels, flat_preds, average='macro', zero_division=0)
    
    def _generate_sample_output(self, batch: Dict) -> Dict[str, Any]:
        """生成样本输出"""
        input_ids = batch["input_ids"][0]
        
        with torch.no_grad():
            generated = self.model.generate(
                input_ids.unsqueeze(0),
                max_length=self.config.max_length,
                num_beams=self.config.num_beams,
                temperature=self.config.temperature,
                top_p=self.config.top_p,
                do_sample=self.config.temperature > 0,
                pad_token_id=self.tokenizer.pad_token_id,
                eos_token_id=self.tokenizer.eos_token_id
            )
        
        input_text = self.tokenizer.decode(input_ids, skip_special_tokens=True)
        output_text = self.tokenizer.decode(generated[0], skip_special_tokens=True)
        
        return {
            "input": input_text[:200],
            "output": output_text[:500]
```

```python
        label_column: Optional[str] = None
    ):
```

```python
class StreamingDataset(IterableDataset):
    """流式数据集"""
    
    def __init__(
        self,
        file_paths: List[str],
        tokenizer,
        max_length: int = 512,
        shuffle_buffer_size: int = 10000
    ):
        self.file_paths = file_paths
        self.tokenizer = tokenizer
        self.max_length = max_length
        self.shuffle_buffer_size = shuffle_buffer_size
    
    def __iter__(self) -> Iterator[Dict]:
        import random
        
        buffer = []
        
        for file_path in self.file_paths:
            with open(file_path, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        item = json.loads(line.strip())
                        
                        encoding = self.tokenizer(
                            item.get("text", ""),
                            max_length=self.max_length,
                            padding="max_length",
                            truncation=True,
                            return_tensors="pt"
                        )
                        
                        processed = {
                            "input_ids": encoding["input_ids"].squeeze(),
                            "attention_mask": encoding["attention_mask"].squeeze(),
                            "labels": encoding["input_ids"].squeeze()
                        }
                        
                        buffer.append(processed)
                        
                        if len(buffer) >= self.shuffle_buffer_size:
                            random.shuffle(buffer)
                            while buffer:
                                yield buffer.pop()
                    except json.JSONDecodeError:
                        continue
        
        random.shuffle(buffer)
        while buffer:
            yield buffer.pop()
```

```python
    async def preprocess_dataset(
        self,
        dataset_id: str,
        operations: List[Dict[str, Any]],
        output_name: str,
        owner_id: str
    ) -> DatasetModel:
        """预处理数据集"""
        source_dataset = await self.get_dataset(dataset_id)
        if not source_dataset:
            raise DatasetNotFoundException(f"Dataset not found: {dataset_id}")
        
        data = await self._load_raw_data(source_dataset.storage_path, source_dataset.format)
        
        for operation in operations:
            op_type = operation.get("type")
            
            if op_type == "filter":
                field = operation["field"]
                condition = operation["condition"]
                value = operation["value"]
                data = self._apply_filter(data, field, condition, value)
            
            elif op_type == "map":
                field = operation["field"]
                function = operation["function"]
                data = self._apply_map(data, field, function)
            
            elif op_type == "sample":
                ratio = operation.get("ratio", 0.1)
                seed = operation.get("seed", 42)
                data = self._apply_sample(data, ratio, seed)
            
            elif op_type == "shuffle":
                seed = operation.get("seed", 42)
                data = self._apply_shuffle(data, seed)
            
            elif op_type == "deduplicate":
                field = operation.get("field")
                data = self._apply_deduplicate(data, field)
        
        new_dataset_id = f"ds_{uuid.uuid4().hex[:12]}"
        output_path = await self.storage.store_processed_data(
            data, new_dataset_id, "jsonl"
        )
        
        new_dataset = DatasetModel(
            dataset_id=new_dataset_id,
            name=output_name,
            description=f"Processed from {source_dataset.name}",
            dataset_type=source_dataset.dataset_type,
            format="jsonl",
            storage_path=output_path,
            size_bytes=os.path.getsize(output_path),
            sample_count=len(data),
            parent_id=dataset_id,
            owner_id=owner_id
        )
        
        self.db.add(new_dataset)
        await self.db.commit()
        
        return new_dataset
    
    def _apply_filter(
        self,
        data: List[Dict],
        field: str,
        condition: str,
        value: Any
    ) -> List[Dict]:
        """应用过滤"""
        if condition == "equals":
            return [d for d in data if d.get(field) == value]
        elif condition == "contains":
            return [d for d in data if value in str(d.get(field, ""))]
        elif condition == "greater_than":
            return [d for d in data if d.get(field, 0) > value]
        elif condition == "less_than":
            return [d for d in data if d.get(field, 0) < value]
        elif condition == "not_empty":
            return [d for d in data if d.get(field)]
        return data
```

```python
    def _apply_map(
        self,
        data: List[Dict],
        field: str,
        function: str
    ) -> List[Dict]:
        """应用映射"""
        result = []
        for item in data:
            new_item = item.copy()
            
            if function == "lowercase":
                if field in new_item:
                    new_item[field] = str(new_item[field]).lower()
            elif function == "uppercase":
                if field in new_item:
                    new_item[field] = str(new_item[field]).upper()
            elif function == "strip":
                if field in new_item:
                    new_item[field] = str(new_item[field]).strip()
            elif function == "truncate_256":
                if field in new_item:
                    new_item[field] = str(new_item[field])[:256]
            
            result.append(new_item)
        
        return result
    
    def _apply_sample(
        self,
        data: List[Dict],
        ratio: float,
        seed: int
    ) -> List[Dict]:
        """应用采样"""
        import random
        random.seed(seed)
        sample_size = int(len(data) * ratio)
        return random.sample(data, sample_size)
    
    def _apply_shuffle(self, data: List[Dict], seed: int) -> List[Dict]:
        """应用打乱"""
        import random
        random.seed(seed)
        shuffled = data.copy()
        random.shuffle(shuffled)
        return shuffled
    
    def _apply_deduplicate(
        self,
        data: List[Dict],
        field: Optional[str]
    ) -> List[Dict]:
        """应用去重"""
        seen = set()
        result = []
        
        for item in data:
            if field:
                key = item.get(field)
            else:
                key = json.dumps(item, sort_keys=True)
            
            if key not in seen:
                seen.add(key)
                result.append(item)
        
        return result
    
    async def _load_raw_data(self, path: str, format: str) -> List[Dict]:
        """加载原始数据"""
        if format == "jsonl":
            data = []
            with open(path, 'r', encoding='utf-8') as f:
                for line in f:
                    try:
                        data.append(json.loads(line.strip()))
                    except:
                        continue
            return data
        elif format == "json":
            with open(path, 'r', encoding='utf-8') as f:
                return json.load(f)
        elif format == "parquet":
            table = pq.read_table(path)
            return table.to_pylist()
        
        return []
```

```python
    async def create_version(
        self,
        model_id: str,
        source_path: str,
        version: Optional[str] = None,
        metadata: Dict[str, Any] = None
    ) -> ModelVersion:
        """创建模型版本"""
        if model_id not in self._models:
            raise ValueError(f"Model not found: {model_id}")
        
        if not version:
            existing_versions = self._versions.get(model_id, [])
            version = f"v{len(existing_versions) + 1}"
        
        version_id = f"{model_id}_{version}"
        
        version_path = os.path.join(self.base_path, model_id, version)
        
        if os.path.isdir(source_path):
            shutil.copytree(source_path, version_path)
        else:
            os.makedirs(version_path, exist_ok=True)
            shutil.copy2(source_path, version_path)
        
        model_hash = await self._compute_model_hash(version_path)
        
        version_metadata = {
            "source_path": source_path,
            "hash": model_hash,
            "size_bytes": self._get_directory_size(version_path),
            **(metadata or {})
        }
        
        model_version = ModelVersion(
            version_id=version_id,
            model_id=model_id,
            version=version,
            path=version_path,
            metadata=version_metadata
        )
        
        self._versions[model_id].append(model_version)
        self._models[model_id]["latest_version"] = version
        
        await self._save_version_metadata(model_version)
        await self._save_model_metadata(model_id)
        
        logger.info(f"Created model version: {version_id}")
        return model_version
    
    async def _compute_model_hash(self, path: str) -> str:
        """计算模型哈希"""
        hasher = hashlib.sha256()
        
        for root, dirs, files in os.walk(path):
            for filename in sorted(files):
                filepath = os.path.join(root, filename)
                with open(filepath, 'rb') as f:
                    while chunk := f.read(8192):
                        hasher.update(chunk)
        
        return hasher.hexdigest()[:16]
    
    def _get_directory_size(self, path: str) -> int:
        """获取目录大小"""
        total = 0
        for root, dirs, files in os.walk(path):
            for filename in files:
                filepath = os.path.join(root, filename)
                total += os.path.getsize(filepath)
        return total
```

```python
class SearchStrategy(str, Enum):
    """搜索策略"""
    GRID = "grid"
    RANDOM = "random"
    BAYESIAN = "bayesian"


@dataclass
class HyperParameter:
    """超参数定义"""
    name: str
    param_type: str
    low: Optional[float] = None
    high: Optional[float] = None
    choices: Optional[List[Any]] = None
    log_scale: bool = False
    
    def sample(self) -> Any:
        """采样"""
        if self.param_type == "float":
            if self.log_scale:
                log_low = math.log(self.low)
                log_high = math.log(self.high)
                return math.exp(random.uniform(log_low, log_high))
            return random.uniform(self.low, self.high)
        
        elif self.param_type == "int":
            return random.randint(int(self.low), int(self.high))
        
        elif self.param_type == "categorical":
            return random.choice(self.choices)
        
        return None


@dataclass
class Trial:

    trial_id: str
    parameters: Dict[str, Any]
    objective_value: Optional[float] = None
```

```python
    async def optimize(self) -> Trial:
        """执行优化"""
        logger.info(f"Starting hyperparameter optimization with {self.strategy}")
        
        if self.strategy == SearchStrategy.GRID:
            await self._grid_search()
        elif self.strategy == SearchStrategy.RANDOM:
            await self._random_search()
        elif self.strategy == SearchStrategy.BAYESIAN:
            await self._bayesian_search()
        
        return self.best_trial
    
    async def _grid_search(self) -> None:
        """网格搜索"""
        import itertools
        
        param_grids = {}
        for name, param in self.search_space.items():
            if param.param_type == "categorical":
                param_grids[name] = param.choices
            elif param.param_type == "int":
                step = max(1, (param.high - param.low) // 5)
                param_grids[name] = list(range(int(param.low), int(param.high) + 1, int(step)))
            else:
                steps = min(5, self.n_trials)
                step = (param.high - param.low) / steps
                param_grids[name] = [param.low + i * step for i in range(steps + 1)]
        
        keys = param_grids.keys()
        combinations = list(itertools.product(*param_grids.values()))
        
        for i, values in enumerate(combinations[:self.n_trials]):
            params = dict(zip(keys, values))
            await self._run_trial(f"trial_{i}", params)
    
    async def _random_search(self) -> None:
        """随机搜索"""
        for i in range(self.n_trials):
            params = {
                name: param.sample()
                for name, param in self.search_space.items()
            }
            await self._run_trial(f"trial_{i}", params)
    
    async def _bayesian_search(self) -> None:
        """贝叶斯优化"""
        try:
            import optuna
            
            def objective(trial):
                params = {}
                for name, param in self.search_space.items():
                    if param.param_type == "float":
                        params[name] = trial.suggest_float(
                            name, param.low, param.high, log=param.log_scale
                        )
                    elif param.param_type == "int":
                        params[name] = trial.suggest_int(name, int(param.low), int(param.high))
                    elif param.param_type == "categorical":
                        params[name] = trial.suggest_categorical(name, param.choices)
                
                loop = asyncio.get_event_loop()
                result = loop.run_until_complete(self.objective_fn(params))
                return result
            
            study = optuna.create_study(
                direction=self.direction,
                sampler=optuna.samplers.TPESampler()
            )
            study.optimize(objective, n_trials=self.n_trials)
            
            for optuna_trial in study.trials:
                trial = Trial(
                    trial_id=f"trial_{optuna_trial.number}",
                    parameters=optuna_trial.params,
                    objective_value=optuna_trial.value,
                    status="completed" if optuna_trial.state.name == "COMPLETE" else "failed"
                )
                self.trials.append(trial)
                self._update_best(trial)
                
        except ImportError:
            logger.warning("Optuna not available, falling back to random search")
            await self._random_search()
```

```python
    async def _run_trial(self, trial_id: str, params: Dict[str, Any]) -> Trial:
        """运行单个试验"""
        trial = Trial(
            trial_id=trial_id,
            parameters=params,
            status="running"
        )
        self.trials.append(trial)
        
        try:
            objective_value = await self.objective_fn(params)
            trial.objective_value = objective_value
            trial.status = "completed"
            
            self._update_best(trial)
            
            logger.info(f"Trial {trial_id}: {objective_value:.4f}")
            
        except Exception as e:
            trial.status = "failed"
            trial.error_message = str(e)
            logger.error(f"Trial {trial_id} failed: {e}")
        
        return trial
    
    def _update_best(self, trial: Trial) -> None:
        """更新最佳试验"""
        if trial.objective_value is None:
            return
        
        if self.best_trial is None:
            self.best_trial = trial
            return
        
        if self.direction == "minimize":
            if trial.objective_value < self.best_trial.objective_value:
                self.best_trial = trial
        else:
            if trial.objective_value > self.best_trial.objective_value:
                self.best_trial = trial
    
    def get_best_params(self) -> Optional[Dict[str, Any]]:
        """获取最佳参数"""
        if self.best_trial:
            return self.best_trial.parameters
        return None
    
    def get_trials_dataframe(self):
        """获取试验数据框"""
        import pandas as pd
        
        data = []
        for trial in self.trials:
            row = {
                "trial_id": trial.trial_id,
                "status": trial.status,
                "objective_value": trial.objective_value,
                **trial.parameters
            }
            data.append(row)
        
        return pd.DataFrame(data)
    
    def plot_optimization_history(self):
        """绘制优化历史"""
        import matplotlib.pyplot as plt
        
        completed_trials = [t for t in self.trials if t.status == "completed"]
        values = [t.objective_value for t in completed_trials]
        
        plt.figure(figsize=(10, 6))
        plt.plot(values, marker='o', label='Objective Value')
        
        best_values = []
        current_best = float('inf') if self.direction == "minimize" else float('-inf')
        for v in values:
            if self.direction == "minimize":
                current_best = min(current_best, v)
            else:
                current_best = max(current_best, v)
            best_values.append(current_best)
        
        plt.plot(best_values, marker='s', label='Best Value')
        plt.xlabel('Trial')
        plt.ylabel('Objective Value')
        plt.title('Hyperparameter Optimization History')
        plt.legend()
        plt.grid(True)
```

```python
    async def _send_notification(self, notification: Notification) -> None:
        """发送通知"""
        tasks = []
        
        for channel in notification.channels:
            config = self._configs.get(channel)
            if not config or not config.enabled:
                continue
            
            if channel == NotificationChannel.EMAIL:
                tasks.append(self._send_email(notification, config))
            elif channel == NotificationChannel.WEBHOOK:
                tasks.append(self._send_webhook(notification, config))
            elif channel == NotificationChannel.DINGTALK:
                tasks.append(self._send_dingtalk(notification, config))
            elif channel == NotificationChannel.FEISHU:
                tasks.append(self._send_feishu(notification, config))
            elif channel == NotificationChannel.SLACK:
                tasks.append(self._send_slack(notification, config))
        
        if tasks:
            results = await asyncio.gather(*tasks, return_exceptions=True)
            for i, result in enumerate(results):
                if isinstance(result, Exception):
                    logger.error(f"Failed to send notification: {result}")
    
    async def _send_email(
        self,
        notification: Notification,
        config: NotificationConfig
    ) -> None:
        """发送邮件"""
        import aiosmtplib
        from email.message import EmailMessage
        
        msg = EmailMessage()
        msg["Subject"] = notification.title
        msg["From"] = config.config.get("from_email")
        msg["To"] = config.config.get("to_emails", [])
        msg.set_content(notification.content)
        
        await aiosmtplib.send(
            msg,
            hostname=config.config.get("smtp_host"),
            port=config.config.get("smtp_port", 587),
            username=config.config.get("smtp_user"),
            password=config.config.get("smtp_password"),
            use_tls=True

                "timestamp": notification.created_at.isoformat()
            }
            
            headers = config.config.get("headers", {})
            
            async with session.post(
                config.config["url"],
```

```python
    async def publish(self, channel: str, message: Dict[str, Any]) -> None:
        """发布消息"""
        if not self._initialized:
            await self.initialize()
        
        await self.redis.publish(channel, json.dumps(message))
    
    async def subscribe(
        self,
        channel: str,
        callback: Callable[[Dict], Any]
    ) -> None:
        """订阅消息"""
        if not self._initialized:
            await self.initialize()
        
        pubsub = self.redis.pubsub()
        await pubsub.subscribe(channel)
        
        self._subscribers[channel] = callback
        
        asyncio.create_task(self._listen(pubsub, channel, callback))
    
    async def _listen(
        self,
        pubsub,
        channel: str,
        callback: Callable
    ) -> None:
        """监听消息"""
        async for message in pubsub.listen():
            if message["type"] == "message":
                try:
                    data = json.loads(message["data"])
                    await callback(data)
                except Exception as e:
                    logger.error(f"Error processing message on {channel}: {e}")
    
    async def unsubscribe(self, channel: str) -> None:
        """取消订阅"""
        if channel in self._subscribers:
            del self._subscribers[channel]
    
    async def close(self) -> None:
        """关闭连接"""
        if self._initialized and self.redis:
```
