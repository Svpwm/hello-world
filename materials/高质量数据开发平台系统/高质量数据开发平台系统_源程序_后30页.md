@celery.task(bind=True, max_retries=3)
def execute_pipeline_task(self, run_id: str):
    """执行管道任务"""
    run = PipelineRun.query.get(run_id)
    if not run:
        logger.error(f'运行记录不存在: {run_id}')
        return
    
    run.start()
    
    try:
        pipeline = run.pipeline
        definition = pipeline.definition
        
        # 解析管道定义
        nodes = definition.get('nodes', [])
        edges = definition.get('edges', [])
        
        # 构建执行顺序
        node_order = build_node_order(nodes, edges)
        
        # 初始化上下文
        context = {
            'params': run.params,
            'variables': pipeline.variables.copy(),
            'outputs': {}
        }
        
        total_records = 0
        success_records = 0
        failed_records = 0
        
        # 按顺序执行节点
        for node_id in node_order:
            node = next((n for n in nodes if n['id'] == node_id), None)
            if not node:
                continue
            
            node_result = execute_node(run.id, node, context)
            context['outputs'][node_id] = node_result
            
            # 累计统计
            total_records += node_result.get('records_in', 0)
            success_records += node_result.get('records_out', 0)
            failed_records += node_result.get('records_failed', 0)
            
            if not node_result.get('success'):
                raise Exception(f"节点执行失败: {node_result.get('error')}")
        
        # 更新统计
        run.records_processed = total_records
        run.records_success = success_records
        run.records_failed = failed_records
        
        run.complete(result=context['outputs'])
        logger.info(f'管道执行成功: {run_id}')
        
    except Exception as e:
        logger.error(f'管道执行失败: {run_id}, 错误: {str(e)}')
        run.fail(str(e))

def build_node_order(nodes: list, edges: list) -> list:
    """构建节点执行顺序（拓扑排序）"""
    graph = {n['id']: [] for n in nodes}
    in_degree = {n['id']: 0 for n in nodes}
    
    for edge in edges:
        source = edge['source']
        target = edge['target']
        graph[source].append(target)
        in_degree[target] += 1
    
    queue = [n_id for n_id, degree in in_degree.items() if degree == 0]
    order = []
    
    while queue:
        current = queue.pop(0)
        order.append(current)
        
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    return order


def execute_node(run_id: str, node: dict, context: dict) -> dict:
    """执行单个节点"""
    node_run = NodeRun(
        run_id=run_id,
        node_id=node['id'],
        input_data=resolve_inputs(node.get('inputs', {}), context),
        status=PipelineRunStatus.RUNNING,
        started_at=datetime.utcnow()
    )
    db.session.add(node_run)
    db.session.commit()
    
    try:
        node_type = node['type']
        config = node.get('config', {})
        inputs = node_run.input_data
        
        # 根据节点类型执行
        if node_type == 'source':
            result = execute_source_node(inputs, config)
        elif node_type == 'transform':
            result = execute_transform_node(inputs, config)
        elif node_type == 'filter':
            result = execute_filter_node(inputs, config)
        elif node_type == 'aggregate':
            result = execute_aggregate_node(inputs, config)
        elif node_type == 'sink':
            result = execute_sink_node(inputs, config)
        else:
            raise ValueError(f'未知的节点类型: {node_type}')

def resolve_inputs(inputs: dict, context: dict) -> dict:
    """解析输入参数"""
    resolved = {}
    
    for key, value in inputs.items():
        if isinstance(value, str) and value.startswith('{{') and value.endswith('}}'):
            var_path = value[2:-2].strip()
            resolved[key] = get_nested_value(context, var_path)
        else:
            resolved[key] = value
    
    return resolved


def get_nested_value(obj: dict, path: str):
    """获取嵌套字典的值"""
    keys = path.split('.')
    value = obj
    for key in keys:
        if isinstance(value, dict):
            value = value.get(key)
        else:
            return None
    return value


def calculate_aggregation(values: list, func: str):
    """计算聚合值"""
    if not values:
        return None
    
    if func == 'count':
        return len(values)
    elif func == 'sum':
        return sum(values)
    elif func == 'avg':
        return sum(values) / len(values)
    elif func == 'min':
        return min(values)
    elif func == 'max':
        return max(values)
    else:
        return None


def eval_condition(condition: str, row: dict) -> bool:
    """评估条件表达式"""
    try:
        return eval(condition, {'__builtins__': {}}, row)
    except:
        return False


def cast_value(value, target_type: str):
    """类型转换"""
    if value is None:
        return None
    
    if target_type == 'int':
        return int(value)
    elif target_type == 'float':
        return float(value)
    elif target_type == 'str':
        return str(value)
    elif target_type == 'bool':
        return bool(value)
    else:
        return value

from functools import wraps
from flask import request, g, jsonify
from marshmallow import ValidationError


def validate_json(schema_class):
    """JSON请求体验证装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            json_data = request.get_json()
            
            if json_data is None:
                return jsonify({
                    'success': False,
                    'error': '请求体必须是JSON格式'
                }), 400
            
            schema = schema_class()
            
            try:
                g.validated_data = schema.load(json_data)
            except ValidationError as e:
                return jsonify({
                    'success': False,
                    'error': '数据验证失败',
                    'details': e.messages
                }), 400
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

"""

from math import ceil
from typing import Any


class Pagination:
    """分页类"""
    
    def __init__(self, query, page: int = 1, per_page: int = 20):
        self.query = query
        self.page = max(1, page)
        self.per_page = min(max(1, per_page), 100)
        
        self.total = query.count()
        self.items = query.offset((self.page - 1) * self.per_page).limit(self.per_page).all()
    
    @property
    def pages(self) -> int:
        """总页数"""
        return ceil(self.total / self.per_page) if self.total > 0 else 1
    
    @property
    def has_prev(self) -> bool:
        """是否有上一页"""
        return self.page > 1
    
    @property
    def has_next(self) -> bool:
        """是否有下一页"""
        return self.page < self.pages

from functools import wraps
from flask import jsonify
from flask_login import current_user


def permission_required(permission):
    """权限验证装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not current_user.can(permission):
                return jsonify({
                    'success': False,
                    'error': '权限不足'
                }), 403
            return f(*args, **kwargs)
        return decorated_function
    return decorator


def rate_limit(limit: int, period: int = 60):
    """请求频率限制装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            from flask import current_app, request
            
            key = f'rate_limit:{request.endpoint}:{current_user.id}'
            redis = current_app.redis
            
            current_count = redis.get(key)
            
            if current_count is None:
                redis.setex(key, period, 1)
            elif int(current_count) >= limit:
                return jsonify({
                    'success': False,
                    'error': '请求过于频繁'
                }), 429
            else:
                redis.incr(key)
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

def cache_response(timeout: int = 300, key_prefix: str = ''):
    """响应缓存装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            from flask import current_app, request
            import hashlib
            import json
            
            cache_key = f'{key_prefix}:{request.path}:{hashlib.md5(request.query_string).hexdigest()}'
            redis = current_app.redis
            
            cached = redis.get(cache_key)
            if cached:
                return jsonify(json.loads(cached))
            
            response = f(*args, **kwargs)
            
            if isinstance(response, tuple):
                data, status_code = response[0], response[1]
            else:
                data, status_code = response, 200
            
            if status_code == 200:
                redis.setex(cache_key, timeout, json.dumps(data.get_json()))
            
            return response
        return decorated_function
    return decorator
