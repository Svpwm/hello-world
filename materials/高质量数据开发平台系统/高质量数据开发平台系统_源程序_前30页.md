def setup_logging(app):
    """配置应用日志"""
    log_level = app.config.get('LOG_LEVEL', 'INFO')
    log_format = '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    
    logging.basicConfig(
        level=getattr(logging, log_level),
        format=log_format
    )
    
    if app.config.get('LOG_FILE'):
        file_handler = logging.FileHandler(app.config['LOG_FILE'])
        file_handler.setFormatter(logging.Formatter(log_format))
        app.logger.addHandler(file_handler)

    def activate(self):
        """激活管道"""
        if not self.definition:
            raise ValueError('管道定义不能为空')
        self.status = PipelineStatus.ACTIVE
        db.session.commit()
    
    def pause(self):
        """暂停管道"""
        self.status = PipelineStatus.PAUSED
        db.session.commit()
    
    def execute(self, params=None):
        """执行管道"""
        if self.status != PipelineStatus.ACTIVE:
            raise ValueError('只有激活状态的管道可以执行')
        
        run = PipelineRun(
            pipeline_id=self.id,
            params=params or {},
            status=PipelineRunStatus.PENDING
        )
        db.session.add(run)
        db.session.commit()
        
        # 触发异步执行
        from app.tasks.pipeline import execute_pipeline
        execute_pipeline.delay(run.id)
        
        return run

    def execute(self):
        """执行质量检查"""

        # 更新统计
        self.check_count += 1

        )

            self._send_alert(result)
        
        return check_result

        elif rule_type == RuleType.VALIDITY:
            return self._check_validity()
        elif rule_type == RuleType.ACCURACY:
            return self._check_accuracy()
        elif rule_type == RuleType.CONSISTENCY:
            return self._check_consistency()
        else:
            raise ValueError(f'不支持的规则类型: {rule_type}')
    
    def _check_completeness(self) -> CheckResult:
        """检查完整性（非空检查）"""
        column = self.rule.column_name
        
        # 获取数据
        data_source = self.dataset.data_source
        connector = DataSourceConnector(data_source)
        
        # 统计总数和非空数
        query = f"""
            SELECT 
                COUNT(*) as total,
                COUNT({column}) as non_null,
                SUM(CASE WHEN {column} IS NULL OR {column} = '' THEN 1 ELSE 0 END) as null_count
            FROM {self.dataset.table_name}
        """
        result = connector.execute_query(query)
        
        if not result:
            return CheckResult(100.0, 0, 0, 0, {})
        
        row = result[0]
        total = row['total']
        valid = row['non_null']
        invalid = row['null_count']
        score = (valid / total * 100) if total > 0 else 100.0
        
        return CheckResult(
            score=round(score, 2),
            total_records=total,
            valid_records=valid,
            invalid_records=invalid,
            details={'column': column, 'check_type': 'completeness'}
        )
